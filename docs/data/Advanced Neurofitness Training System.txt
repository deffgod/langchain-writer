2. Apply progressive complexity scaling
3. Balance training modalities (strength, cardio, neuromotor)
4. Include recovery periods

2. Monitor fatigue and recovery markers
3. Adjust intensity based on user feedback
4. Track progress metrics

### 1.1 Основные компоненты системы

   - Классификация по целям

3. **Генератор программ**
   - Подбор упражнений

   - Адаптация нагрузок

    type: "Основная тренировка";

    type: "Дополнительная тренировка";
    duration: number;
    videos: string[];
  };
  recovery: {
    type: "Восстановление";
    duration: number;
    videos: string[];

const createTrainingDay = (
  goal: string,
  intensity: "low" | "medium" | "high",
  userPreferences: UserPreferences

  const mapping = goalMappings[goal];

    morning: intensity === "high" ? {
      type: "Нейрозарядка",
      duration: 20,
      videos: selectVideos(mapping.videoSelection.warmup)
    } : undefined,
    main: {
      type: "Основная тренировка",

      type: "Дополнительная тренировка",

  recoveryDays: string[];

    modifyExercises?: boolean;
    addComplexity?: boolean;
  };
}




const createWeeklyPlan = (
  weekNumber: number,
  userProfile: UserProfile,
  previousWeek?: WeeklyPlan
): WeeklyPlan =&gt; {
  const intensityDistribution = calculateIntensityDistribution(weekNumber, userProfile);
  const trainingDays = {};
  
  userProfile.availableDays.forEach(day =&gt; {
    trainingDays[day] = createTrainingDay(
      userProfile.goal,
      intensityDistribution[day],
      userProfile.preferences
    );
  });




  return {
    weekNumber,
    focus: determineFocus(weekNumber, userProfile.goal),
    intensityDistribution,
    trainingDays,
    recoveryDays: calculateRecoveryDays(userProfile.availableDays),

```sql
-- Create search function for videos
CREATE OR REPLACE FUNCTION content.search_videos(
    search_query TEXT,
    filter_categories TEXT[] DEFAULT NULL,
    filter_effects TEXT[] DEFAULT NULL,
    page INTEGER DEFAULT 1,
    page_size INTEGER DEFAULT 20
) RETURNS TABLE (
    video_id VARCHAR(50),
    name VARCHAR(255),
    description TEXT,
    duration VARCHAR(10),
    trainer_name VARCHAR(100),
    categories TEXT[],
    effects TEXT[],
    equipment TEXT[],
    total_likes INTEGER,
    total_views INTEGER,
    rank FLOAT
)
LANGUAGE plpgsql
AS $$
BEGIN
    RETURN QUERY
    WITH search_results AS (
        SELECT
            v.video_id,
            v.name,
            v.description,
            v.duration,
            v.trainer_name,
            ARRAY_AGG(DISTINCT c.name) FILTER (WHERE c.name IS NOT NULL) AS categories,
            ARRAY_AGG(DISTINCT ne.name) FILTER (WHERE ne.name IS NOT NULL) AS effects,
            ARRAY_AGG(DISTINCT e.name) FILTER (WHERE e.name IS NOT NULL) AS equipment,
            COALESCE(vm.likes, 0) AS total_likes,
            COALESCE(vm.views, 0) AS total_views,
            ts_rank(v.search_vector, to_tsquery('russian', search_query)) AS rank
        FROM
            content.videos v
        LEFT JOIN
            content.video_categories vc ON v.video_id = vc.video_id
        LEFT JOIN
            content.categories c ON vc.category_id = c.category_id
        LEFT JOIN
            content.video_effects ve ON v.video_id = ve.video_id
        LEFT JOIN
            content.neuro_effects ne ON ve.effect_id = ne.effect_id
        LEFT JOIN
            content.video_equipment vq ON v.video_id = vq.video_id
        LEFT JOIN
            content.equipment e ON vq.equipment_id = e.equipment_id
        LEFT JOIN
            metrics.video_metrics vm ON v.video_id = vm.video_id
        WHERE
            (v.search_vector @@ to_tsquery('russian', search_query)
            OR v.name ILIKE '%' || search_query || '%')
            AND (filter_categories IS NULL OR c.name = ANY(filter_categories))
            AND (filter_effects IS NULL OR ne.name = ANY(filter_effects))
        GROUP BY
            v.video_id, v.name, v.description, v.duration, v.trainer_name,
            v.search_vector, vm.likes, vm.views
    )
    SELECT
        sr.video_id,
        sr.name,
        sr.description,
        sr.duration,
        sr.trainer_name,
        sr.categories,
        sr.effects,
        sr.equipment,
        sr.total_likes,
        sr.total_views,
        sr.rank
    FROM
        search_results sr
    ORDER BY
        sr.rank DESC, sr.total_likes DESC
    LIMIT page_size
    OFFSET (page - 1) * page_size;
END;
$$;
```

## Backup and Recovery Procedures

```sql
-- Create function for scheduled database backups
CREATE OR REPLACE FUNCTION maintenance.backup_database()
RETURNS VOID AS $$
DECLARE
    backup_path TEXT := '/var/backups/neurofitness/';
    backup_filename TEXT := 'neurofitness_backup_' || to_char(CURRENT_TIMESTAMP, 'YYYY_MM_DD_HH24_MI_SS') || '.sql';
    full_path TEXT;
BEGIN
    -- Create backup directory if it doesn't exist
    EXECUTE format('CREATE DIRECTORY IF NOT EXISTS %s', backup_path);

    full_path := backup_path || backup_filename;

    -- Execute pg_dump
    EXECUTE format('pg_dump --dbname=neurofitness --format=custom --file=%s', full_path);

    -- Log the backup operation
    INSERT INTO maintenance.backup_log (
        backup_time,
        backup_path,
        backup_type,
        status,
        notes
    )
    VALUES (
        CURRENT_TIMESTAMP,
        full_path,
        'FULL',
        'SUCCESS',
        'Scheduled backup completed successfully'
    );

    -- Cleanup old backups (keep only last 7 days)
    EXECUTE 'DELETE FROM maintenance.backup_log WHERE backup_time < CURRENT_TIMESTAMP - INTERVAL ''7 days''';

    EXCEPTION WHEN OTHERS THEN
        -- Log backup failure
        INSERT INTO maintenance.backup_log (
            backup_time,
            backup_path,
            backup_type,
            status,
            notes
        )
        VALUES (
            CURRENT_TIMESTAMP,
            full_path,
            'FULL',
            'FAILED',
            'Backup failed: ' || SQLERRM
        );

        RAISE;
END;
$$ LANGUAGE plpgsql;

-- Schedule daily backups
SELECT cron.schedule('0 1 * * *', 'SELECT maintenance.backup_database()');
```

-- Create function for database maintenance
CREATE OR REPLACE FUNCTION maintenance.perform_maintenance()
RETURNS VOID AS $$
BEGIN
    -- Vacuum analyze all tables
    VACUUM ANALYZE;

    -- Refresh materialized views
    PERFORM refresh_materialized_views();

    -- Update database statistics
    ANALYZE;

    -- Log maintenance operation
    INSERT INTO maintenance.maintenance_log (
        maintenance_type,
        details,
        status
    )
    VALUES (
        'ROUTINE',
        'Vacuum analyze, refresh materialized views, and update statistics',
        'SUCCESS'
    );

    EXCEPTION WHEN OTHERS THEN
        -- Log maintenance failure
        INSERT INTO maintenance.maintenance_log (
            maintenance_type,
            details,
            status,
            error_message
        )
        VALUES (
            'ROUTINE',
            'Vacuum analyze, refresh materialized views, and update statistics',
            'FAILED',
            SQLERRM
        );

        RAISE;
END;
$$ LANGUAGE plpgsql;

# Database Schema Documentation

This comprehensive SQL database design for the Neurofitness System provides the following key features:
1. **Schema Organization**:
   - User data schema: Stores all user profiles, biometrics, preferences, and goals
   - Content schema: Manages the video library, categorization, and effects
   - Training schema: Handles fitness plans, schedules, and workouts
   - Metrics schema: Tracks user progress and engagement metrics
   - Maintenance schema: Supports database health and backup operations
2. **Relational Structure**:
   - Properly normalized tables with clear relationships
   - Foreign key constraints for data integrity
   - Unique constraints to prevent duplicates
3. **Security Features**:
   - Role-based access control
   - Row-level security policies for sensitive data
   - Secure authentication context management
4. **Performance Optimization**:
   - Indexed columns for faster queries
   - Materialized views for complex aggregations
   - Full-text search capabilities
5. **Data Maintenance**:
   - Scheduled backups
   - Automatic database maintenance
   - Data import/export functions
6. **Business Logic**:
   - Stored procedures for common operations
   - Automatic data updates via triggers
   - Standardized error handling

This database design efficiently organizes the neurofitness data while maintaining proper relationships between users, courses, videos, and training plans. It provides a solid foundation for building a comprehensive neurofitness application.
