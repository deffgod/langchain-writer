        """Verify user access requirements"""

    def generate_plan(self, user_profile):

        if not self.validate_access(user_profile):

        selected_workouts = self.select_workouts(user_profile, neuro_results)

        weekly_plan = []
        training_days = user_profile['planSettings']['trainingDays']

        for week in range(user_profile['planSettings']['programDurationWeeks']):
            weekly_structure = {
                "weekNumber": week + 1,

            for day in training_days:

                weekly_structure["workouts"].append(daily_workouts)

        daily_workouts = []

                    "category": workout['category'],
                    "duration": workout['duration'],
                    "trainer": workout['trainer'],
                    "description": workout['description']

                remaining_duration -= workout['duration']

        return {

class NeuroPlanParser:
    def __init__(self):
        self.heart_rate_zones = {
            'recovery': (0.5, 0.6),
            'aerobic': (0.6, 0.7),
            'anaerobic': (0.7, 0.8),
            'maximum': (0.8, 0.9)
        }

    def calculate_heart_rate_zones(self, max_hr, resting_hr):
        """
        Karvonen Formula for Heart Rate Zones
        THR = ((HRmax - HRrest) × % Intensity) + HRrest
        """
        zones = {}
        heart_rate_reserve = max_hr - resting_hr

        for zone_name, (lower, upper) in self.heart_rate_zones.items():
            zones[zone_name] = {
                'lower': round(heart_rate_reserve * lower + resting_hr),
                'upper': round(heart_rate_reserve * upper + resting_hr)
            }
        return zones

    def parse_neuro_test_results(self, test_results):

        available_days = user_profile.get('trainingDays', [])
        workout_distribution = self.distribute_workouts(workouts, len(available_days))

        schedule = {}
        for day, day_workouts in zip(available_days, workout_distribution):
            schedule[day] = {
                'workouts': day_workouts,
                'total_duration': sum(w.get('duration', 0) for w in day_workouts),
                'neural_load': sum(self.calculate_neural_load(w) for w in day_workouts)
            }

        return schedule

    def distribute_workouts(workouts, days):
        """Distribute workouts across available days"""
        distributed = [[] for _ in range(days)]
        workout_index = 0

        for workout in workouts:
            distributed[workout_index % days].append(workout)
            workout_index += 1

        return distributed

    if hours &gt; 0:
        return f"{hours}ч {mins}мин"
    return f"{mins}мин"

def calculate_training_load(workout_data):
    """Calculate training load using Duration x Intensity"""
    intensity_multipliers = {
        'low': 1,
        'medium': 2,
        'high': 3
    }
    return workout_data['duration'] * intensity_multipliers.get(workout_data['intensity'], 1)

def generate_workout_id():
    """Generate unique workout ID"""
    import uuid
    return f"WO_{uuid.uuid4().hex[:8]}"
